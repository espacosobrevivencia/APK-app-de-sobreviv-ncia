@CapacitorPlugin(
    name = "CentralBridge",
    permissions = {
        @Permission(alias = "location", strings = { Manifest.permission.ACCESS_COARSE_LOCATION, Manifest.permission.ACCESS_FINE_LOCATION }),
        @Permission(alias = "coarseLocation", strings = { Manifest.permission.ACCESS_COARSE_LOCATION })
    }
)
public class CentralBridgePlugin extends Plugin {
    private static final String PREFS = "central_native";
    private static final String FOLDERS = "folders";

    private JSObject locationPermissionResult() {
        PermissionState precise = getPermissionState("location");
        PermissionState coarse = getPermissionState("coarseLocation");
        String preciseText = precise == null ? "prompt" : precise.toString();
        String coarseText = coarse == null ? "prompt" : coarse.toString();
        String state;
        if (precise == PermissionState.GRANTED) state = "granted";
        else if (coarse == PermissionState.GRANTED) state = "approximate";
        else if (precise != null) state = precise.toString();
        else state = "prompt";
        JSObject ret = new JSObject();
        ret.put("state", state);
        ret.put("precise", preciseText);
        ret.put("coarse", coarseText);
        return ret;
    }

    @PluginMethod
    public void checkLocationPermission(PluginCall call) {
        call.resolve(locationPermissionResult());
    }

    @PluginMethod
    public void requestLocationPermission(PluginCall call) {
        PermissionState precise = getPermissionState("location");
        PermissionState coarse = getPermissionState("coarseLocation");
        if (precise == PermissionState.GRANTED || coarse == PermissionState.GRANTED) {
            call.resolve(locationPermissionResult());
            return;
        }
        requestPermissionForAlias("location", call, "locationPermissionCallback");
    }

    @PermissionCallback
    private void locationPermissionCallback(PluginCall call) {
        call.resolve(locationPermissionResult());
    }

    @PluginMethod
    public void ping(PluginCall call) {
        JSObject ret = new JSObject();
        ret.put("ok", true);
        ret.put("platform", "android");
        call.resolve(ret);
    }

    @PluginMethod
    public void isAppInstalled(PluginCall call) {
        String packageName = call.getString("packageName", "");
        boolean installed = false;
        if (!packageName.isEmpty()) {
            try {
                installed = getContext().getPackageManager().getLaunchIntentForPackage(packageName) != null;
            } catch (Exception ignored) {}
        }
        JSObject ret = new JSObject(); ret.put("installed", installed); call.resolve(ret);
    }

    @PluginMethod
    public void openApp(PluginCall call) {
        String packageName = call.getString("packageName", "");
        try {
            Intent launch = getContext().getPackageManager().getLaunchIntentForPackage(packageName);
            if (launch == null) {
                launch = new Intent(Intent.ACTION_MAIN);
                launch.addCategory(Intent.CATEGORY_LAUNCHER);
                launch.setPackage(packageName);
            }
            launch.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
            getContext().startActivity(launch);
            JSObject r=new JSObject();r.put("opened",true);call.resolve(r);
        } catch (Exception e) { JSObject r=new JSObject();r.put("opened",false);call.resolve(r); }
    }

    @PluginMethod
    public void openSystemAction(PluginCall call) {
        String action = call.getString("action", "");
        Intent intent;
        if ("camera".equals(action)) intent = new Intent("android.media.action.IMAGE_CAPTURE");
        else if ("recorder".equals(action)) intent = new Intent("android.provider.MediaStore.RECORD_SOUND");
        else { JSObject r=new JSObject();r.put("opened",false);call.resolve(r);return; }
        try { intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK); getContext().startActivity(intent); JSObject r=new JSObject();r.put("opened",true);call.resolve(r); }
        catch(Exception e){ JSObject r=new JSObject();r.put("opened",false);call.resolve(r); }
    }

    @PluginMethod
    public void getBattery(PluginCall call) {
        BatteryManager bm=(BatteryManager)getContext().getSystemService(Context.BATTERY_SERVICE);
        int level=bm!=null?bm.getIntProperty(BatteryManager.BATTERY_PROPERTY_CAPACITY):-1;
        JSObject r=new JSObject();r.put("level",level);call.resolve(r);
    }


    @PluginMethod
    public void getPowerState(PluginCall call) {
        BatteryManager bm=(BatteryManager)getContext().getSystemService(Context.BATTERY_SERVICE);
        int level=bm!=null?bm.getIntProperty(BatteryManager.BATTERY_PROPERTY_CAPACITY):-1;
        Intent battery=getContext().registerReceiver(null,new IntentFilter(Intent.ACTION_BATTERY_CHANGED));
        int status=battery!=null?battery.getIntExtra(BatteryManager.EXTRA_STATUS,-1):-1;
        int plugged=battery!=null?battery.getIntExtra(BatteryManager.EXTRA_PLUGGED,0):0;
        boolean charging=status==BatteryManager.BATTERY_STATUS_CHARGING||status==BatteryManager.BATTERY_STATUS_FULL;
        String pluggedName="";
        if(plugged==BatteryManager.BATTERY_PLUGGED_USB)pluggedName="USB";
        else if(plugged==BatteryManager.BATTERY_PLUGGED_AC)pluggedName="carregador";
        else if(plugged==BatteryManager.BATTERY_PLUGGED_WIRELESS)pluggedName="sem fio";
        PowerManager pm=(PowerManager)getContext().getSystemService(Context.POWER_SERVICE);
        JSObject r=new JSObject();
        r.put("level",level);
        r.put("charging",charging);
        r.put("plugged",pluggedName);
        r.put("powerSaveMode",pm!=null&&pm.isPowerSaveMode());
        call.resolve(r);
    }

    @PluginMethod
    public void openPowerSettings(PluginCall call) {
        String action=call.getString("action","");
        Intent i;
        if("saver".equals(action))i=new Intent(Settings.ACTION_BATTERY_SAVER_SETTINGS);
        else if("battery-usage".equals(action)||"usage".equals(action))i=new Intent(Settings.ACTION_BATTERY_USAGE_SETTINGS);
        else if("battery-optimization".equals(action))i=new Intent(Settings.ACTION_IGNORE_BATTERY_OPTIMIZATION_SETTINGS);
        else if("usage-access".equals(action))i=new Intent(Settings.ACTION_USAGE_ACCESS_SETTINGS);
        else if("display".equals(action))i=new Intent(Settings.ACTION_DISPLAY_SETTINGS);
        else if("wireless".equals(action))i=new Intent(Settings.ACTION_WIRELESS_SETTINGS);
        else if("bluetooth".equals(action))i=new Intent(Settings.ACTION_BLUETOOTH_SETTINGS);
        else if("location".equals(action))i=new Intent(Settings.ACTION_LOCATION_SOURCE_SETTINGS);
        else {JSObject r=new JSObject();r.put("opened",false);call.resolve(r);return;}
        try{i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);getContext().startActivity(i);JSObject r=new JSObject();r.put("opened",true);call.resolve(r);}
        catch(Exception e){JSObject r=new JSObject();r.put("opened",false);call.resolve(r);}
    }

    @PluginMethod
    public void openAppPowerSettings(PluginCall call) {
        String packageName=call.getString("packageName","").trim();
        if(packageName.isEmpty()){JSObject r=new JSObject();r.put("opened",false);call.resolve(r);return;}
        try{
            Intent i=new Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS,Uri.parse("package:"+packageName));
            i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
            getContext().startActivity(i);
            JSObject r=new JSObject();r.put("opened",true);call.resolve(r);
        }catch(Exception e){JSObject r=new JSObject();r.put("opened",false);call.resolve(r);}
    }

    private boolean usageAccessGranted() {
        try{
            AppOpsManager appOps=(AppOpsManager)getContext().getSystemService(Context.APP_OPS_SERVICE);
            if(appOps==null)return false;
            int mode=appOps.checkOpNoThrow(AppOpsManager.OPSTR_GET_USAGE_STATS, Process.myUid(), getContext().getPackageName());
            return mode==AppOpsManager.MODE_ALLOWED;
        }catch(Exception e){return false;}
    }

    @PluginMethod
    public void hasUsageAccess(PluginCall call) {
        JSObject r=new JSObject();r.put("granted",usageAccessGranted());call.resolve(r);
    }

    @PluginMethod
    public void getUsageStats(PluginCall call) {
        JSArray out=new JSArray();
        if(!usageAccessGranted()){JSObject r=new JSObject();r.put("apps",out);r.put("granted",false);call.resolve(r);return;}
        Integer h=call.getInt("hours");int hours=h==null?24:Math.max(1,Math.min(168,h));
        long end=System.currentTimeMillis(),start=end-hours*60L*60L*1000L;
        try{
            UsageStatsManager manager=(UsageStatsManager)getContext().getSystemService(Context.USAGE_STATS_SERVICE);
            List<UsageStats> stats=manager==null?new ArrayList<>():manager.queryUsageStats(UsageStatsManager.INTERVAL_DAILY,start,end);
            if(stats==null)stats=new ArrayList<>();
            Collections.sort(stats,(a,b)->Long.compare(b.getTotalTimeInForeground(),a.getTotalTimeInForeground()));
            int count=0;
            for(UsageStats s:stats){
                if(s==null||s.getTotalTimeInForeground()<=0)continue;
                JSObject x=new JSObject();x.put("packageName",s.getPackageName());x.put("foregroundMs",s.getTotalTimeInForeground());x.put("lastTimeUsed",s.getLastTimeUsed());out.put(x);
                if(++count>=50)break;
            }
            JSObject r=new JSObject();r.put("apps",out);r.put("granted",true);call.resolve(r);
        }catch(Exception e){JSObject r=new JSObject();r.put("apps",out);r.put("granted",true);call.resolve(r);}
    }

    @PluginMethod
    public void pickFolder(PluginCall call) {
        Intent intent = new Intent(Intent.ACTION_OPEN_DOCUMENT_TREE);
        intent.addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION | Intent.FLAG_GRANT_PERSISTABLE_URI_PERMISSION | Intent.FLAG_GRANT_PREFIX_URI_PERMISSION);
        startActivityForResult(call, intent, "folderPicked");
    }

    @ActivityCallback
    private void folderPicked(PluginCall call, ActivityResult result) {
        if (call == null) return;
        if (result.getResultCode() != Activity.RESULT_OK || result.getData() == null || result.getData().getData() == null) {
            call.reject("Seleção cancelada"); return;
        }
        Uri uri = result.getData().getData();
        int flags = result.getData().getFlags() & (Intent.FLAG_GRANT_READ_URI_PERMISSION | Intent.FLAG_GRANT_WRITE_URI_PERMISSION);
        int takeFlags = flags & Intent.FLAG_GRANT_READ_URI_PERMISSION;
        try { if (takeFlags != 0) getContext().getContentResolver().takePersistableUriPermission(uri, takeFlags); } catch(Exception ignored) {}
        SharedPreferences prefs=getContext().getSharedPreferences(PREFS,Context.MODE_PRIVATE);
        Set<String> folders=new HashSet<>(prefs.getStringSet(FOLDERS,new HashSet<>()));folders.add(uri.toString());prefs.edit().putStringSet(FOLDERS,folders).apply();
        JSObject ret=new JSObject();ret.put("uri",uri.toString());ret.put("name",queryDisplayName(uri));call.resolve(ret);
    }

    private String queryDisplayName(Uri uri) {
        Cursor c=null; try { c=getContext().getContentResolver().query(uri,new String[]{DocumentsContract.Document.COLUMN_DISPLAY_NAME},null,null,null);if(c!=null&&c.moveToFirst())return c.getString(0); } catch(Exception ignored) {} finally { if(c!=null)c.close(); }
        return "Pasta Android";
    }

    @PluginMethod
    public void listFolder(PluginCall call) {
        String uriString=call.getString("uri",""); Integer requested=call.getInt("limit"); int limit=requested==null?8000:Math.max(1,Math.min(8000,requested));
        if(uriString.isEmpty()){call.reject("URI ausente");return;}
        try {
            Uri tree=Uri.parse(uriString);String rootId=DocumentsContract.getTreeDocumentId(tree);JSArray files=new JSArray();int[] count={0};boolean[] limited={false};
            walkTree(tree,rootId,"",files,count,limit,limited);
            JSObject ret=new JSObject();ret.put("files",files);ret.put("count",count[0]);ret.put("limited",limited[0]);call.resolve(ret);
        } catch(Exception e){call.reject("Não foi possível listar a pasta",e);}
    }

    private void walkTree(Uri tree,String parentId,String path,JSArray out,int[] count,int limit,boolean[] limited){
        if(count[0]>=limit){limited[0]=true;return;}
        Uri children=DocumentsContract.buildChildDocumentsUriUsingTree(tree,parentId);
        String[] cols={DocumentsContract.Document.COLUMN_DOCUMENT_ID,DocumentsContract.Document.COLUMN_DISPLAY_NAME,DocumentsContract.Document.COLUMN_MIME_TYPE,DocumentsContract.Document.COLUMN_SIZE,DocumentsContract.Document.COLUMN_LAST_MODIFIED};
        Cursor c=null;
        try {
            c=getContext().getContentResolver().query(children,cols,null,null,null);
            if(c==null)return;
            while(c.moveToNext()){
                if(count[0]>=limit){limited[0]=true;return;}
                String docId=c.getString(0),name=c.getString(1),mime=c.getString(2);long size=c.isNull(3)?0:c.getLong(3),last=c.isNull(4)?0:c.getLong(4);String rel=path.isEmpty()?name:path+"/"+name;
                if(DocumentsContract.Document.MIME_TYPE_DIR.equals(mime)){walkTree(tree,docId,rel,out,count,limit,limited);if(limited[0])return;}
                else {Uri doc=DocumentsContract.buildDocumentUriUsingTree(tree,docId);JSObject f=new JSObject();f.put("uri",doc.toString());f.put("name",name);f.put("mime",mime==null?"":mime);f.put("size",size);f.put("lastModified",last);f.put("relativePath",rel);out.put(f);count[0]++;}
            }
        } catch(Exception ignored) {} finally { if(c!=null)c.close(); }
    }

    @PluginMethod
    public void openDocument(PluginCall call) {
        String uriString=call.getString("uri","");String mime=call.getString("mime","*/*");
        try {Intent i=new Intent(Intent.ACTION_VIEW);i.setDataAndType(Uri.parse(uriString),mime==null||mime.isEmpty()?"*/*":mime);i.addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION|Intent.FLAG_ACTIVITY_NEW_TASK);getContext().startActivity(i);JSObject r=new JSObject();r.put("opened",true);call.resolve(r);}catch(Exception e){JSObject r=new JSObject();r.put("opened",false);call.resolve(r);}
    }

    @PluginMethod
    public void shareDocument(PluginCall call) {
        String uriString=call.getString("uri","");String mime=call.getString("mime","*/*");
        try {Intent i=new Intent(Intent.ACTION_SEND);i.setType(mime==null||mime.isEmpty()?"*/*":mime);i.putExtra(Intent.EXTRA_STREAM,Uri.parse(uriString));i.addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION|Intent.FLAG_ACTIVITY_NEW_TASK);Intent chooser=Intent.createChooser(i,"Compartilhar arquivo");chooser.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);getContext().startActivity(chooser);JSObject r=new JSObject();r.put("shared",true);call.resolve(r);}catch(Exception e){JSObject r=new JSObject();r.put("shared",false);call.resolve(r);}
    }
}
